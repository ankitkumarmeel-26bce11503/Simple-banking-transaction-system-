balance = 5000
def check_balance():
    print("\nYour current balance is: ₹", balance)


def deposit_money():
    global balance
    amount = float(input("\nEnter the amount you want to deposit: ₹"))
    if amount > 0:
        balance = balance + amount
        print("₹", amount, "has been deposited successfully.")
        print("Updated balance: ₹", balance)
    else:
        print("Please enter a valid amount.")


def withdraw_money():
    global balance
    amount = float(input("\nEnter the amount you want to withdraw: ₹"))
    if amount <= 0:
        print("Please enter a valid amount.")
    elif amount > balance:
        print("Insufficient balance.")
        print("Your current balance is: ₹", balance)
    else:
        balance = balance - amount
        print("₹", amount, "has been withdrawn successfully.")
        print("Remaining balance: ₹", balance)




def banking_system():


    print("====================================")
    print("       SIMPLE BANKING SYSTEM")
    print("====================================")


    while True:


        print("\n----------- MENU -----------")
        print("1. Check Balance")
        print("2. Deposit Money")
        print("3. Withdraw Money")
        print("4. Exit")
        print("----------------------------")


        choice = input("Enter your choice: ")


        if choice == "1":
            check_balance()


        elif choice == "2":
            deposit_money()


        elif choice == "3":
            withdraw_money()


        elif choice == "4":
            print("\nThank you for using our banking system.")
            print("Have a nice day!")
            break
        else:
            print("\nInvalid choice. Please try again.")




# Start the banking system
banking_system()
